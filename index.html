export default { async fetch(request) { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") { return new Response(null, { headers: cors }); } try { const url = new URL(request.url); const stock = url.searchParams.get("stock"); if (!stock) { return new Response( JSON.stringify({ error: "請提供股票代號" }), { status: 400, headers: { ...cors, "Content-Type": "application/json" } } ); } const twseHost = "mis.twse.com.tw"; const apiUrl = "https://" + twseHost + "/stock/api/getStockInfo.jsp" + "?ex_ch=tse_" + encodeURIComponent(stock) + ".tw" + "&json=1" + "&delay=0"; const response = await fetch(apiUrl, { headers: { "User-Agent": "Mozilla/5.0", "Accept": "application/json" } }); const data = await response.json(); return new Response( JSON.stringify(data), { headers: { ...cors, "Content-Type": "application/json; charset=utf-8" } } ); } catch (error) { return new Response( JSON.stringify({ error: error.message }), { status: 500, headers: { ...cors, "Content-Type": "application/json" } } ); } } };
